		static	void move_b(t_stack **a, t_stack **b, int len_a)
{
	while (len_a-- > 3)
	{
		index_list(a);
		index_list(b);
		get_above_median(*a);
		get_above_median(*b);
		set_target_b(*a, *b);
		push_cost(*a, *b);
		move_node(a, b, find_cheapest(*a));
	}
}

static	void move_a(t_stack **b, t_stack **a)
{
	int		len_b;

	len_b = list_size_circular(*b);
	while (len_b-- > 0)
	{
		index_list(a);
		index_list(b);
		get_above_median(*a);
		get_above_median(*b);
		set_target_a(*b, *a);
		push_cost(*a, *b);
		inverse_move_node(b, a, find_cheapest(*b));
	}
}
